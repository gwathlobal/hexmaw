; A cave cut by a 3 m band of lava: a running jump, or a 45-degree causeway.
; Its plan is drawn by tools/cave_plans.py.
cells 3 3
tags room cave lava
weight 1
mirror yes
material rock 0.36 0.32 0.3
material cave-floor 0.4 0.35 0.3
remap wall rock
remap ceiling rock
remap floor cave-floor
legend
  ~ lava
  ( diagonal / floor -0.4 0, lava    ; a shore: floor on one half, lava on the other
  ) diagonal \ floor -0.4 0, lava
  < diagonal /, lava                 ; 45-degree rock standing in lava
  > diagonal \, lava
  i spawn any north
  $ pickup floor
  & pickup reward

plan 0
